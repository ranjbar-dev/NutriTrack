**`frontend/app/stores/labResult.ts`**:
```typescript
import { defineStore } from 'pinia'
import type { LabResultResponse, LabType } from '~/types/tracking.types'
import { LAB_TYPE_LABELS } from '~/types/tracking.types'

export const useLabResultStore = defineStore('labResult', () => {
  const labResults = ref<LabResultResponse[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  async function fetchLabResults(clientId?: string) {
    loading.value = true
    try {
      const { apiFetch } = useApi()
      const url = clientId
        ? `/nutritionist/clients/${clientId}/lab-results`
        : '/client/lab-results'
      labResults.value = await apiFetch<LabResultResponse[]>(url)
    } catch (e: unknown) {
      error.value = (e as { data?: { error?: string } }).data?.error ?? 'خطا در بارگذاری'
    } finally {
      loading.value = false
    }
  }

  async function uploadLabResult(formData: FormData) {
    const { apiFetch } = useApi()
    // IMPORTANT: pass FormData as body — do NOT set Content-Type header
    // apiFetch must detect FormData and omit Content-Type so browser sets multipart boundary
    const result = await apiFetch<LabResultResponse>('/client/lab-results', {
      method: 'POST',
      body: formData,
    })
    labResults.value.unshift(result)  // newest first
  }

  async function downloadLabResult(clientId: string, labResultId: string) {
    // Authenticated download: fetch with auth header, create blob URL
    const { apiFetch } = useApi()
    // Use raw fetch with responseType blob
    const { $fetch } = useNuxtApp()
    const blob = await $fetch<Blob>(
      `/api/nutritionist/clients/${clientId}/lab-results/${labResultId}/download`,
      { responseType: 'blob' }
    )
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `lab-result-${labResultId}`
    document.body.appendChild(a)
    a.click()
    document.body.removeChild(a)
    URL.revokeObjectURL(url)
  }

  function $reset() { labResults.value = []; error.value = null }
  return { labResults, loading, error, fetchLabResults, uploadLabResult, downloadLabResult, $reset }
})

export { LAB_TYPE_LABELS }
```

**`frontend/app/components/tracking/LabResultUploadForm.vue`**:
```vue
<script setup lang="ts">
import type { LabType } from '~/types/tracking.types'
import { LAB_TYPE_LABELS } from '~/types/tracking.types'

const emit = defineEmits<{ (e: 'uploaded'): void }>()
const labStore = useLabResultStore()

const selectedLabType = ref<LabType | ''>('')
const notes = ref('')
const fileInput = ref<HTMLInputElement | null>(null)
const uploading = ref(false)
const formError = ref<string | null>(null)

const labTypeOptions = Object.entries(LAB_TYPE_LABELS) as [LabType, string][]

async function submit() {
  if (!selectedLabType.value) { formError.value = 'نوع آزمایش الزامی است'; return }
  if (!fileInput.value?.files?.length) { formError.value = 'فایل آزمایش الزامی است'; return }
  formError.value = null
  uploading.value = true
  try {
    const fd = new FormData()
    fd.append('lab_type', selectedLabType.value)
    fd.append('notes', notes.value)
    fd.append('local_id', crypto.randomUUID())   // TRACK-12
    fd.append('file', fileInput.value.files[0])
    await labStore.uploadLabResult(fd)
    selectedLabType.value = ''
    notes.value = ''
    if (fileInput.value) fileInput.value.value = ''
    emit('uploaded')
  } catch (e: unknown) {
    formError.value = (e as { data?: { error?: string } }).data?.error ?? 'خطا در آپلود فایل'
  } finally {
    uploading.value = false
  }
}
</script>

<template>
  <div class="bg-white rounded-2xl shadow-sm p-4 mb-4">
    <h2 class="text-sm font-semibold text-gray-700 mb-3 text-start">بارگذاری آزمایش</h2>
    <div class="mb-3">
      <label class="block text-xs text-gray-500 mb-1 text-start">نوع آزمایش *</label>
      <select v-model="selectedLabType" class="w-full border rounded-xl p-3 text-start bg-white">
        <option value="" disabled>انتخاب کنید...</option>
        <option v-for="[type, label] in labTypeOptions" :key="type" :value="type">{{ label }}</option>
      </select>
    </div>
    <div class="mb-3">
      <label class="block text-xs text-gray-500 mb-1 text-start">توضیحات (اختیاری)</label>
      <textarea v-model="notes" rows="2" class="w-full border rounded-xl p-3 text-start resize-none" placeholder="مثلاً: ناشتا انجام شد" />
    </div>
    <div class="mb-3">
      <label class="block text-xs text-gray-500 mb-1 text-start">فایل آزمایش * (تصویر یا PDF)</label>
      <input ref="fileInput" type="file" accept="image/*,.pdf"
        class="w-full border rounded-xl p-3 text-start file:bg-green-50 file:text-green-700 file:border-0 file:rounded-lg" />
    </div>
    <p v-if="formError" class="text-red-500 text-sm mb-3 text-start">{{ formError }}</p>
    <button @click="submit" :disabled="uploading"
      class="w-full bg-green-500 text-white py-3 rounded-xl font-medium disabled:opacity-50">
      {{ uploading ? 'در حال آپلود...' : 'آپلود آزمایش' }}
    </button>
  </div>
</template>
```
