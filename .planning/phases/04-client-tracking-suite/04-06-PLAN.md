**`frontend/app/stores/exerciseLog.ts`**:
```typescript
import { defineStore } from 'pinia'
import type { ExerciseLogEntry, LogExercisePayload } from '~/types/tracking.types'

export const useExerciseLogStore = defineStore('exerciseLog', () => {
  const todayLogs = ref<ExerciseLogEntry[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  async function fetchToday(date?: string) {
    loading.value = true
    try {
      const { apiFetch } = useApi()
      const today = date ?? new Date().toISOString().slice(0, 10)
      todayLogs.value = await apiFetch<ExerciseLogEntry[]>(`/client/exercise-logs?date=${today}`)
    } catch (e: unknown) {
      error.value = (e as { data?: { error?: string } }).data?.error ?? 'خطا در بارگذاری'
    } finally {
      loading.value = false
    }
  }

  async function logExercise(payload: Omit<LogExercisePayload, 'local_id' | 'date'>) {
    const { apiFetch } = useApi()
    const body: LogExercisePayload = {
      local_id: crypto.randomUUID(),   // TRACK-12
      date: new Date().toISOString().slice(0, 10),
      ...payload,
    }
    const entry = await apiFetch<ExerciseLogEntry>('/client/exercise-logs', {
      method: 'POST',
      body: JSON.stringify(body),
    })
    todayLogs.value.push(entry)
  }

  function $reset() { todayLogs.value = []; error.value = null }
  return { todayLogs, loading, error, fetchToday, logExercise, $reset }
})
```

**`frontend/app/stores/medicationLog.ts`** (from 04-RESEARCH.md Pattern 5):
```typescript
import { defineStore } from 'pinia'
import type { MedicationLogEntry, LogMedicationPayload } from '~/types/tracking.types'

export const useMedicationLogStore = defineStore('medicationLog', () => {
  const todayLogs = ref<MedicationLogEntry[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  // Checklist derived from active plan (D-18, Pattern 5) — no extra API call
  const checklistItems = computed(() => {
    const planStore = useClientPlanStore()
    const plan = planStore.activePlan
    if (!plan) return []
    return plan.medications.flatMap(med =>
      (med.times ?? []).map(time => ({
        prescribedMedicationId: med.id,
        medicationName: med.medication_name,
        dosage: med.dosage ?? null,
        time,
        isTaken: todayLogs.value.some(
          l => l.prescribed_medication_id === med.id && l.taken_at?.slice(0, 5) === time
        ),
      }))
    )
  })

  async function fetchToday(date?: string) {
    loading.value = true
    try {
      const { apiFetch } = useApi()
      const today = date ?? new Date().toISOString().slice(0, 10)
      todayLogs.value = await apiFetch<MedicationLogEntry[]>(`/client/medication-logs?date=${today}`)
    } catch (e: unknown) {
      error.value = (e as { data?: { error?: string } }).data?.error ?? 'خطا در بارگذاری'
    } finally {
      loading.value = false
    }
  }

  async function logMedication(payload: Omit<LogMedicationPayload, 'local_id' | 'date'>) {
    const { apiFetch } = useApi()
    const body: LogMedicationPayload = {
      local_id: crypto.randomUUID(),   // TRACK-12
      date: new Date().toISOString().slice(0, 10),
      ...payload,
    }
    const entry = await apiFetch<MedicationLogEntry>('/client/medication-logs', {
      method: 'POST',
      body: JSON.stringify(body),
    })
    // Optimistic local update — no plan re-fetch (Pitfall 6)
    todayLogs.value.push(entry)
  }

  function $reset() { todayLogs.value = []; error.value = null }
  return { todayLogs, checklistItems, loading, error, fetchToday, logMedication, $reset }
})
```
