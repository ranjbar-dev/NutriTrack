**`frontend/app/stores/bodyMeasurement.ts`**:
```typescript
import { defineStore } from 'pinia'
import type { BodyMeasurementEntry, LogBodyMeasurementPayload } from '~/types/tracking.types'

export const useBodyMeasurementStore = defineStore('bodyMeasurement', () => {
  const history = ref<BodyMeasurementEntry[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  async function fetchHistory(clientId?: string) {
    loading.value = true
    try {
      const { apiFetch } = useApi()
      // Client route (no clientId): own history; Nutritionist route passes clientId
      const url = clientId
        ? `/nutritionist/clients/${clientId}/tracking/body`
        : '/client/body-measurements'
      history.value = await apiFetch<BodyMeasurementEntry[]>(url)
    } catch (e: unknown) {
      error.value = (e as { data?: { error?: string } }).data?.error ?? 'خطا در بارگذاری'
    } finally {
      loading.value = false
    }
  }

  async function logMeasurement(payload: Omit<LogBodyMeasurementPayload, 'local_id' | 'date'>) {
    const { apiFetch } = useApi()
    const body: LogBodyMeasurementPayload = {
      local_id: crypto.randomUUID(),   // TRACK-12
      date: new Date().toISOString().slice(0, 10),
      ...payload,
    }
    await apiFetch<BodyMeasurementEntry>('/client/body-measurements', {
      method: 'POST',
      body: JSON.stringify(body),
    })
    // Re-fetch to reflect last-write-wins server state (TRACK-11)
    await fetchHistory()
  }

  function $reset() { history.value = []; error.value = null }
  return { history, loading, error, fetchHistory, logMeasurement, $reset }
})
```

<script setup lang="ts">
const emit = defineEmits<{ (e: 'submitted'): void }>()
const bodyStore = useBodyMeasurementStore()

const form = reactive({
  weight_kg: '',
  waist_cm: '',
  hip_cm: '',
  chest_cm: '',
  body_fat_pct: '',
})
const showExtra = ref(false)
const saving = ref(false)
const formError = ref<string | null>(null)

async function submit() {
  if (!form.weight_kg || Number(form.weight_kg) <= 0) {
    formError.value = 'وزن الزامی است'
    return
  }
  formError.value = null
  saving.value = true
  try {
    await bodyStore.logMeasurement({
      weight_kg: Number(form.weight_kg),
      waist_cm: form.waist_cm ? Number(form.waist_cm) : undefined,
      hip_cm: form.hip_cm ? Number(form.hip_cm) : undefined,
      chest_cm: form.chest_cm ? Number(form.chest_cm) : undefined,
      body_fat_pct: form.body_fat_pct ? Number(form.body_fat_pct) : undefined,
    })
    Object.assign(form, { weight_kg: '', waist_cm: '', hip_cm: '', chest_cm: '', body_fat_pct: '' })
    showExtra.value = false
    emit('submitted')
  } catch (e: unknown) {
    formError.value = (e as { data?: { error?: string } }).data?.error ?? 'خطا در ذخیره'
  } finally {
    saving.value = false
  }
}
</script>

<template>
