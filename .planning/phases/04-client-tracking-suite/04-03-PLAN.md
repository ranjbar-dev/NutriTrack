```go
func (s *TrackingService) CreateLabResult(ctx context.Context, clientID uuid.UUID, req dto.CreateLabResultRequest, fileReader io.Reader, fileSize int64, originalFilename string) (*dto.LabResultResponse, error) {
    var filePath, origFilename, mimeTypeStr *string
    var sizeBytes *int64

    if fileReader != nil {
        // Validate file size (must be checked before reading — use fileSize from multipart header)
        if fileSize > 10<<20 {
            return nil, ErrLabFileTooLarge
        }

        // Detect MIME type from magic bytes (D-26)
        kind, err := mimetype.DetectReader(fileReader)
        if err != nil || !isAllowedMIME(kind.String()) {
            return nil, ErrLabFileInvalidType
        }

        // Build file path: UPLOADS_DIR/lab-results/{clientID}/{uuid}.ext
        ext := extensionFromMIME(kind.String()) // ".pdf", ".jpg", ".png"
        uuidName := uuid.New().String() + ext
        relPath := filepath.Join("lab-results", clientID.String(), uuidName)
        absPath := filepath.Join(s.uploadsDir, relPath)

        if err := os.MkdirAll(filepath.Dir(absPath), 0755); err != nil {
            return nil, fmt.Errorf("failed to create upload dir: %w", err)
        }

        // Seek back to start after MIME detection (DetectReader reads first 512 bytes)
        if seeker, ok := fileReader.(io.Seeker); ok {
            seeker.Seek(0, io.SeekStart)
        }

        dst, err := os.Create(absPath)
        if err != nil {
            return nil, fmt.Errorf("failed to create file: %w", err)
        }
        defer dst.Close()
        if _, err := io.Copy(dst, fileReader); err != nil {
            return nil, fmt.Errorf("failed to save file: %w", err)
        }

        // Store relative path in DB (Pitfall 3 — never store absolute path)
        fp := relPath
        fn := originalFilename
        mt := kind.String()
        sb := fileSize
        filePath = &fp
        origFilename = &fn
        mimeTypeStr = &mt
        sizeBytes = &sb
    }

    // LAB-02: at least one of file or link required
    if filePath == nil && (req.ExternalLink == nil || *req.ExternalLink == "") {
        return nil, ErrLabFileMissing
    }

    return s.repo.CreateLabResult(ctx, repository.CreateLabResultParams{
        ClientID:         clientID,
        UploadedBy:       clientID, // uploaded by the client themselves
        Title:            req.Title,
        LabType:          req.LabType,
        TestDate:         req.TestDate,
        FilePath:         filePath,
        ExternalLink:     req.ExternalLink,
        OriginalFilename: origFilename,
        MimeType:         mimeTypeStr,
        FileSizeBytes:    sizeBytes,
    })
}

func isAllowedMIME(mime string) bool {
    return mime == "application/pdf" || mime == "image/jpeg" || mime == "image/png"
}

func extensionFromMIME(mime string) string {
    switch mime {
    case "application/pdf":  return ".pdf"
    case "image/jpeg":       return ".jpg"
    case "image/png":        return ".png"
    default:                 return ".bin"
    }
}
```

```go
    clientID := uuid.MustParse(c.GetString("user_id"))

    if err := c.Request.ParseMultipartForm(10 << 20); err != nil {
        c.JSON(http.StatusBadRequest, dto.ErrorResponse{Error: "فایل بیش از حد مجاز است"})
        return
    }

    var req dto.CreateLabResultRequest
    if err := c.ShouldBind(&req); err != nil {
        c.JSON(http.StatusBadRequest, dto.ErrorResponse{Error: "اطلاعات ورودی نامعتبر است"})
        return
    }

    var fileReader io.Reader
    var fileSize int64
    var originalFilename string
    if fh, err := c.FormFile("file"); err == nil {
        f, _ := fh.Open()
        defer f.Close()
        fileReader = f
        fileSize = fh.Size
        originalFilename = fh.Filename
    }

    resp, err := h.svc.CreateLabResult(c.Request.Context(), clientID, req, fileReader, fileSize, originalFilename)
    if err != nil {
        switch {
        case errors.Is(err, service.ErrLabFileMissing):
            c.JSON(http.StatusBadRequest, dto.ErrorResponse{Error: err.Error()})
        case errors.Is(err, service.ErrLabFileInvalidType):
            c.JSON(http.StatusBadRequest, dto.ErrorResponse{Error: err.Error()})
        case errors.Is(err, service.ErrLabFileTooLarge):
            c.JSON(http.StatusBadRequest, dto.ErrorResponse{Error: err.Error()})
        default:
            c.JSON(http.StatusInternalServerError, dto.ErrorResponse{Error: "خطای داخلی سرور"})
        }
        return
    }
    c.JSON(http.StatusCreated, resp)
}
```

```go
    nutritionistID := uuid.MustParse(c.GetString("user_id"))

    labID := uuid.MustParse(c.Param("labId"))
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client browser → API | client_id for all writes sourced from JWT, never from request body |
| Multipart file upload → Filesystem | File content validated via magic bytes before any filesystem write |
| File download → Browser | Content-Disposition: attachment prevents inline rendering and XSS via SVG/HTML |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-03-01 | Elevation of Privilege | UploadLabResult client_id | mitigate | `clientID = uuid.MustParse(c.GetString("user_id"))` — JWT only, never body field |
| T-04-03-02 | Tampering | File MIME spoofing | mitigate | `mimetype.DetectReader()` magic bytes before persist; Content-Type header never trusted |
| T-04-03-03 | Denial of Service | Oversized file upload | mitigate | `c.Request.ParseMultipartForm(10 << 20)` rejects >10MB before reading file bytes |
| T-04-03-04 | Information Disclosure | File path traversal | mitigate | Download handler uses stored relative path from DB + joins with `cfg.UploadsDir`; user never supplies file path |
| T-04-03-05 | XSS | Inline file rendering | mitigate | `Content-Disposition: attachment` always set; MIME enforcement blocks non-PDF/JPG/PNG at upload |
| T-04-03-06 | Elevation of Privilege | Nutritionist cross-access | mitigate | All nutritionist read methods pass `nutritionistID` from JWT to repository SQL JOIN |
</threat_model>
