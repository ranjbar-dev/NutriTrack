**`frontend/app/composables/useNutriTracking.ts`**:

Shared composable used by all 6 domain sub-pages. Provides date range state and a typed fetch wrapper.

```typescript
import type {
  FoodLogEntry, WaterLogEntry, SleepLogEntry,
  ExerciseLogEntry, MedicationLogEntry, BodyMeasurementEntry,
} from '~/types/tracking.types'

type DomainMap = {
  food: FoodLogEntry[]
  water: WaterLogEntry[]
  sleep: SleepLogEntry[]
  exercise: ExerciseLogEntry[]
  medication: MedicationLogEntry[]
  body: BodyMeasurementEntry[]
}

export function useNutriTracking(clientId: Ref<string>) {
  const { apiFetch } = useApi()

  // Default date range: last 14 days
  const today = new Date()
  const twoWeeksAgo = new Date(today)
  twoWeeksAgo.setDate(today.getDate() - 14)

  const dateTo = ref(today.toISOString().slice(0, 10))
  const dateFrom = ref(twoWeeksAgo.toISOString().slice(0, 10))

  async function fetchDomain<K extends keyof DomainMap>(domain: K): Promise<DomainMap[K]> {
    return apiFetch<DomainMap[K]>(
      `/nutritionist/clients/${clientId.value}/tracking/${domain}?from=${dateFrom.value}&to=${dateTo.value}`
    )
  }

  return { dateFrom, dateTo, fetchDomain }
}
```

**`sleep.vue`**: fetchDomain('sleep'). Show sleep_time, wake_time, computed duration (same midnight-crossing formula as 04-05).

```vue
<script setup lang="ts">
import type { SleepLogEntry } from '~/types/tracking.types'
definePageMeta({ middleware: ['auth'] })
const route = useRoute()
const clientId = computed(() => route.params.clientId as string)
const { dateFrom, dateTo, fetchDomain } = useNutriTracking(clientId)
const items = ref<SleepLogEntry[]>([])
const loading = ref(false)
async function load() {
  loading.value = true
  try { items.value = await fetchDomain('sleep') } finally { loading.value = false }
}
onMounted(load)
watch([dateFrom, dateTo], load)

function durationLabel(entry: SleepLogEntry): string {
  const [sh, sm] = entry.sleep_time.split(':').map(Number)
  const [wh, wm] = entry.wake_time.split(':').map(Number)
  let diff = (wh * 60 + wm) - (sh * 60 + sm)
  if (diff < 0) diff += 1440
  const h = Math.floor(diff / 60)
  const m = diff % 60
  return h > 0 ? `${h} ساعت ${m > 0 ? m + ' دقیقه' : ''}` : `${m} دقیقه`
}
</script>
<template>
  <div class="px-4 pt-4 pb-20">
    <div class="flex gap-2 mb-4">
      <input type="date" v-model="dateFrom" @change="load" class="flex-1 border rounded-xl p-2 text-sm" />
      <span class="self-center text-gray-400 text-sm">تا</span>
      <input type="date" v-model="dateTo" @change="load" class="flex-1 border rounded-xl p-2 text-sm" />
    </div>
    <div v-if="loading" class="text-center text-gray-400 py-6 text-sm">در حال بارگذاری...</div>
    <div v-else-if="items.length === 0" class="text-center text-gray-400 py-6 text-sm">داده‌ای موجود نیست</div>
    <div v-for="item in items" :key="item.id" class="bg-white rounded-xl shadow-sm p-3 mb-2">
      <div class="flex justify-between mb-1">
        <span class="text-gray-800 text-sm">{{ item.sleep_time }} — {{ item.wake_time }}</span>
        <span class="text-xs text-gray-400">{{ item.date }}</span>
      </div>
      <p class="text-xs text-green-700">{{ durationLabel(item) }}</p>
    </div>
  </div>
</template>
```
