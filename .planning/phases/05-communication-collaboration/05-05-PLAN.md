### 3. Polling composable

**File**: `frontend/app/composables/useMessagePolling.ts`

```typescript
export function useMessagePolling(partnerID: Ref<string>) {
  const store = useMessageStore()
  let timer: ReturnType<typeof setInterval> | null = null

  function start() {
    stop()
    timer = setInterval(() => store.pollNewMessages(partnerID.value), 10_000)
  }

  function stop() {
    if (timer) { clearInterval(timer); timer = null }
  }

  onMounted(start)
  onUnmounted(stop)

  return { start, stop }
}
```
