// PushPayload mirrors backend dto.PushPayload
interface PushPayload {
  type: string
  title: string
  body: string
  url?: string
  icon?: string
}

self.addEventListener('push', (event: PushEvent) => {
  if (!event.data) return

  let payload: PushPayload
  try {
    payload = event.data.json() as PushPayload
  }
  catch {
    // Malformed push data — show a generic notification
    payload = {
      type: 'generic',
      title: 'نوتری‌ترک',
      body: 'اعلان جدید',
    }
  }

  event.waitUntil(
    self.registration.showNotification(payload.title, {
      body: payload.body,
      icon: payload.icon ?? '/icons/icon-192x192.png',
      badge: '/icons/badge-72x72.png',
      dir: 'rtl',
      lang: 'fa',
      tag: payload.type,          // same-type notifications replace each other
      renotify: false,            // do not vibrate again for same tag
      data: {
        url: payload.url ?? '/client',
      },
    }),
  )
})

```typescript
self.addEventListener('notificationclick', (event: NotificationEvent) => {
  event.notification.close()

  const targetUrl: string = (event.notification.data as { url?: string })?.url ?? '/client'

  event.waitUntil(
    (self as unknown as ServiceWorkerGlobalScope).clients
      .matchAll({ type: 'window', includeUncontrolled: true })
      .then((windowClients) => {
        // If app window is already open, focus it and navigate
        const existing = windowClients.find(
          c => c.url.includes(targetUrl) && 'focus' in c,
        )
        if (existing) {
          return (existing as WindowClient).focus().then(w => w.navigate(targetUrl))
        }
        // Open a new window
        return (self as unknown as ServiceWorkerGlobalScope).clients.openWindow(targetUrl)
      }),
  )
})
```

```typescript
// Handles browser push permission request and backend subscription registration.
// D-16: Callers must invoke this AFTER the active plan loads — not on app startup.
// This is a plain .ts composable (not .client.ts) but uses browser APIs guarded by typeof window checks.

import { ref } from 'vue'
import { useApi } from '~/composables/useApi'

export type PermissionStatus = 'idle' | 'requesting' | 'subscribed' | 'denied' | 'ios-not-installed' | 'error'

export function useNotificationPermission() {
  const status = ref<PermissionStatus>('idle')
  const errorMessage = ref<string | null>(null)

  // Convert VAPID public key from base64url to Uint8Array (required by pushManager.subscribe)
  function urlB64ToUint8Array(base64String: string): Uint8Array {
    const padding = '='.repeat((4 - (base64String.length % 4)) % 4)
    const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/')
    const rawData = atob(base64)
    const outputArray = new Uint8Array(rawData.length)
    for (let i = 0; i < rawData.length; ++i) {
      outputArray[i] = rawData.charCodeAt(i)
    }
    return outputArray
  }

  async function requestAndSubscribe(): Promise<void> {
    if (typeof window === 'undefined' || !('serviceWorker' in navigator) || !('PushManager' in window)) {
      status.value = 'error'
      errorMessage.value = 'مرورگر شما از اعلان‌ها پشتیبانی نمی‌کند'
      return
    }

    // D-16: iOS detection — must be installed as PWA before push subscription
    const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent)
    const isStandalone = window.matchMedia('(display-mode: standalone)').matches
      || ('standalone' in window.navigator && (window.navigator as { standalone?: boolean }).standalone === true)

    if (isIOS && !isStandalone) {
      status.value = 'ios-not-installed'
      errorMessage.value = 'برای دریافت اعلان، ابتدا اپ را از دکمه «افزودن به صفحه اصلی» نصب کنید'
      return
    }

    // Request notification permission
    status.value = 'requesting'
    const permission = await Notification.requestPermission()

    if (permission !== 'granted') {
      status.value = 'denied'
      errorMessage.value = 'مجوز اعلان داده نشد. از تنظیمات مرورگر می‌توانید آن را فعال کنید.'
      return
    }

    try {
      const registration = await navigator.serviceWorker.ready
      const vapidPublicKey = import.meta.env.VITE_VAPID_PUBLIC_KEY as string

      if (!vapidPublicKey) {
        throw new Error('VITE_VAPID_PUBLIC_KEY not configured')
      }

      const subscription = await registration.pushManager.subscribe({
        userVisibleOnly: true,
        applicationServerKey: urlB64ToUint8Array(vapidPublicKey),
      })

      const sub = subscription.toJSON() as {
        endpoint: string
        keys: { p256dh: string; auth: string }
      }

      // POST subscription to backend
      const { apiFetch } = useApi()
      await apiFetch('/push/subscribe', {
        method: 'POST',
        body: JSON.stringify({
          endpoint: sub.endpoint,
          keys: { p256dh: sub.keys.p256dh, auth: sub.keys.auth },
        }),
      })

      status.value = 'subscribed'
    }
    catch (e: unknown) {
      status.value = 'error'
      const err = e as { message?: string }
      errorMessage.value = err?.message ?? 'خطا در فعال‌سازی اعلان‌ها'
    }
  }

  return { status, errorMessage, requestAndSubscribe }
}
```

**Create frontend/app/stores/notificationPrefs.ts** (D-22):

```typescript
import { ref } from 'vue'
import { defineStore } from 'pinia'
import { useApi } from '~/composables/useApi'
import { db } from '~/db'

export interface NotificationPrefs {
  new_message: boolean
  plan_activated: boolean
  food_request_decision: boolean
  meal_reminder: boolean
  medication_reminder: boolean
  water_reminder: boolean
}

const DEFAULT_PREFS: NotificationPrefs = {
  new_message: true,
  plan_activated: true,
  food_request_decision: true,
  meal_reminder: true,
  medication_reminder: true,
  water_reminder: false,
}

export const useNotificationPrefsStore = defineStore('notificationPrefs', () => {
  const prefs = ref<NotificationPrefs>({ ...DEFAULT_PREFS })
  const loading = ref(false)
  const saving = ref(false)
  const error = ref<string | null>(null)

  async function fetchPreferences(): Promise<void> {
    // D-22 Step 1: Read from Dexie cache first
    try {
      const cached = await db.notificationPreferences.get(1)
      if (cached?.data) {
        prefs.value = cached.data as NotificationPrefs
      }
    }
    catch { /* Dexie unavailable */ }

    if (typeof navigator !== 'undefined' && !navigator.onLine) return

    loading.value = true
    error.value = null
    try {
      const { apiFetch } = useApi()
      const data = await apiFetch<NotificationPrefs>('/push/preferences')
      prefs.value = data
      // Persist to Dexie
      await db.notificationPreferences.put({ id: 1, data, updated_at: new Date().toISOString() })
    }
    catch (e: unknown) {
      const err = e as { data?: { error?: string } }
      if (!prefs.value) {
        error.value = err?.data?.error ?? 'خطا در دریافت تنظیمات اعلان'
      }
    }
    finally {
      loading.value = false
    }
  }

  async function updatePreferences(updates: Partial<NotificationPrefs>): Promise<void> {
    // Optimistic update
    prefs.value = { ...prefs.value, ...updates }
    saving.value = true
    error.value = null
    try {
      const { apiFetch } = useApi()
      const updated = await apiFetch<NotificationPrefs>('/push/preferences', {
        method: 'PATCH',
        body: JSON.stringify(prefs.value),
      })
      prefs.value = updated
      await db.notificationPreferences.put({ id: 1, data: updated, updated_at: new Date().toISOString() })
    }
    catch (e: unknown) {
      const err = e as { data?: { error?: string } }
      error.value = err?.data?.error ?? 'خطا در ذخیره تنظیمات'
      // Revert optimistic update
      await fetchPreferences()
    }
    finally {
      saving.value = false
    }
  }

  return { prefs, loading, saving, error, fetchPreferences, updatePreferences }
})
```
