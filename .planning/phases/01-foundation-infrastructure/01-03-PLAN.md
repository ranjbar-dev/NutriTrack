       type KavenegarSender struct {
           apiKey   string

           client   *http.Client
       }

       func NewKavenegarSender(apiKey, template string) *KavenegarSender {
           return &KavenegarSender{
               apiKey:   apiKey,
               template: template,
               client:   &http.Client{Timeout: 10 * time.Second},
           }
       }

       func (k *KavenegarSender) SendOTP(phone, code string) error {
           // Kavenegar verify/lookup API
           url := fmt.Sprintf(
               "https://api.kavenegar.com/v1/%s/verify/lookup.json?receptor=%s&token=%s&template=%s",
               k.apiKey, phone, code, k.template,
           )

           if err != nil {
               return fmt.Errorf("kavenegar request failed: %w", err)
           }
           defer resp.Body.Close()
           if resp.StatusCode != 200 {
               body, _ := io.ReadAll(resp.Body)
               return fmt.Errorf("kavenegar returned %d: %s", resp.StatusCode, string(body))
           }
           return nil
       }

    1. Create `backend/internal/middleware/auth.go` per D-01 and RESEARCH.md Pattern 2:
       ```go
       func Auth(jwtSecret []byte) gin.HandlerFunc {
           return func(c *gin.Context) {
               tokenStr, err := c.Cookie("access_token")
               if err != nil {
                   c.AbortWithStatusJSON(401, gin.H{"error": "احراز هویت الزامی است"})
                   return
               }
               claims, err := jwt.ParseToken(tokenStr, jwtSecret)
               if err != nil {
                   c.AbortWithStatusJSON(401, gin.H{"error": "توکن نامعتبر است"})
                   return
               }
               // Set user context for downstream handlers
               c.Set("user_id", claims.UserID)
               c.Set("role", claims.Role)
               c.Next()
           }
       }
       ```
       **IMPORTANT per D-01:** Token is extracted from httpOnly cookie named "access_token", NOT from Authorization header. The cookie path is "/api" so it's sent only to API routes.

    2. Create `backend/internal/middleware/role_guard.go`:
       ```go
       func RoleGuard(allowedRoles ...string) gin.HandlerFunc {
           return func(c *gin.Context) {
               role := c.GetString("role")
               for _, allowed := range allowedRoles {
                   if role == allowed {
                       c.Next()
                       return
                   }
               }
               c.AbortWithStatusJSON(403, gin.H{"error": "دسترسی غیرمجاز"})
           }
       }
       ```
       This middleware MUST be applied after Auth middleware (it reads "role" from context set by Auth).

    2. Create `backend/internal/middleware/cors.go` per D-24 and SEC-06:
       ```go
       func CORS(frontendURL string) gin.HandlerFunc {
           return func(c *gin.Context) {
               c.Header("Access-Control-Allow-Origin", frontendURL)
               c.Header("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
               c.Header("Access-Control-Allow-Headers", "Content-Type, Authorization")
               c.Header("Access-Control-Allow-Credentials", "true")
               c.Header("Access-Control-Max-Age", "86400")

               if c.Request.Method == "OPTIONS" {
                   c.AbortWithStatus(204)
                   return
               }
               c.Next()
           }
       }
       ```
       **CRITICAL:** `Allow-Credentials: true` required for httpOnly cookies. Origin MUST be explicit URL (not `*`).

    3. Create `backend/internal/middleware/request_id.go`:
       ```go
       func RequestID() gin.HandlerFunc {
           return func(c *gin.Context) {
               requestID := uuid.New().String()
               c.Set("request_id", requestID)
               c.Header("X-Request-ID", requestID)
               c.Next()
           }
       }
       ```

    4. Create `backend/internal/middleware/logger.go` per D-23 and INFRA-04:
       ```go
       func Logger(logger zerolog.Logger) gin.HandlerFunc {
           return func(c *gin.Context) {
               start := time.Now()
               c.Next()
               duration := time.Since(start)

               logger.Info().
                   Str("request_id", c.GetString("request_id")).
                   Str("method", c.Request.Method).
                   Str("path", c.Request.URL.Path).
                   Int("status", c.Writer.Status()).
                   Dur("duration_ms", duration).
                   Str("client_ip", c.ClientIP()).
                   Msg("request")
           }
       }
       ```
       Fields per D-23: timestamp (zerolog adds automatically), level, method, path, status, duration_ms, request_id.
       Output: structured JSON to stdout for Loki collection.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → API | Untrusted requests; JWT in cookie authenticates |
| API → External SMS | Kavenegar REST API; API key in env var |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Spoofing | JWT | mitigate | Validate signing method is HMAC only in ParseToken (prevent algorithm confusion) |
| T-03-02 | Elevation of Privilege | OTP brute force | mitigate | Rate limiter 3 req/10min + phone normalization (SEC-07, Pitfall 3) |
| T-03-03 | Tampering | CORS bypass | mitigate | Explicit origin, no wildcard, credentials=true (SEC-06) |
| T-03-04 | Information Disclosure | SMS API key | mitigate | Loaded from env var, never logged or exposed in responses |
| T-03-05 | Spoofing | Token in cookie | mitigate | httpOnly + Secure + SameSite cookie flags (D-01) |
| T-03-06 | Repudiation | API requests | mitigate | zerolog structured logging with request_id for traceability (INFRA-04) |
</threat_model>
