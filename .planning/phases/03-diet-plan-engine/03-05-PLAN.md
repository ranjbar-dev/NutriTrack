**2. `plans/new.vue`** — Create new plan:
- Form fields: start_date (Shamsi text input, placeholder "۱۴۰۳/۰۱/۰۱"), end_date (optional), notes (textarea), daily_water_target_ml (number input, default 2000)
- On submit: parse Shamsi input → jalaali.toGregorian → format as "YYYY-MM-DD" → POST /api/diet-plans
- On success: navigate to `/nutritionist/clients/${clientId}/plans/${newPlan.id}`
- Validation: start_date required; show Persian field-level errors
- Use Breadcrumb at top

```vue
<script setup lang="ts">
import jalaali from 'jalaali-js'
import { usePlanBuilderStore } from '~/stores/planBuilder'
import type { DietPlanResponse } from '~/types/plan.types'
const route = useRoute()
const router = useRouter()
const clientId = route.params.clientId as string
const store = usePlanBuilderStore()
const submitting = ref(false)
const error = ref<string | null>(null)

const form = reactive({
  startDateShamsi: '',
  endDateShamsi: '',
  notes: '',
  waterMl: 2000
})

function parseShamsi(input: string): string | null {
  // Accepts YYYY/MM/DD format (Persian or Latin digits)
  const normalized = input.replace(/[۰-۹]/g, d => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
  const parts = normalized.split('/')
  if (parts.length !== 3) return null
  const [jy, jm, jd] = parts.map(Number)
  if (isNaN(jy) || isNaN(jm) || isNaN(jd)) return null
  const { gy, gm, gd } = jalaali.toGregorian(jy, jm, jd)
  return `${gy}-${String(gm).padStart(2,'0')}-${String(gd).padStart(2,'0')}`
}

async function onSubmit() {
  error.value = null
  const startDate = parseShamsi(form.startDateShamsi)
  if (!startDate) { error.value = 'تاریخ شروع معتبر نیست'; return }
  const endDate = form.endDateShamsi ? parseShamsi(form.endDateShamsi) : undefined
  submitting.value = true
  try {
    const plan = await store.createPlan(clientId, {
      start_date: startDate,
      end_date: endDate,
      notes: form.notes || undefined,
      daily_water_target_ml: form.waterMl || undefined,
    }) as DietPlanResponse
    router.push(`/nutritionist/clients/${clientId}/plans/${plan.id}`)
  } catch (e: any) {
    error.value = e.data?.error ?? 'خطا در ایجاد برنامه'
  } finally { submitting.value = false }
}
</script>
<template>
  <div class="min-h-screen bg-gray-50" dir="rtl">
    <div class="sticky top-0 bg-white z-10 px-4 py-3 border-b border-gray-100">
      <PlanBreadcrumb :items="[{label:'برنامه‌ها', to:`/nutritionist/clients/${clientId}/plans`},{label:'برنامه جدید'}]" />
    </div>
    <form @submit.prevent="onSubmit" class="p-4 space-y-4 max-w-lg mx-auto">
      <div v-if="error" class="bg-red-50 text-red-700 rounded-xl px-4 py-3 text-sm">{{ error }}</div>

      <div>
        <label class="block text-sm font-medium text-gray-700 mb-1">تاریخ شروع <span class="text-red-500">*</span></label>
        <input v-model="form.startDateShamsi" type="text" placeholder="۱۴۰۳/۰۱/۰۱"
          class="w-full bg-white border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500" />
      </div>
      <div>
        <label class="block text-sm font-medium text-gray-700 mb-1">تاریخ پایان (اختیاری)</label>
        <input v-model="form.endDateShamsi" type="text" placeholder="۱۴۰۳/۰۱/۰۷"
          class="w-full bg-white border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500" />
      </div>
      <div>
        <label class="block text-sm font-medium text-gray-700 mb-1">یادداشت (اختیاری)</label>
        <textarea v-model="form.notes" rows="3"
          class="w-full bg-white border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500 resize-none" />
      </div>
      <div>
        <label class="block text-sm font-medium text-gray-700 mb-1">هدف آب روزانه (میلی‌لیتر)</label>
        <input v-model.number="form.waterMl" type="number" min="0" step="100"
          class="w-full bg-white border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500" />
      </div>
      <button type="submit" :disabled="submitting"
        class="w-full bg-emerald-600 text-white py-3 rounded-xl font-medium hover:bg-emerald-700 disabled:opacity-50 transition-colors">
        {{ submitting ? 'در حال ذخیره...' : 'ایجاد برنامه' }}
      </button>
    </form>
  </div>
</template>
```

<script setup lang="ts">
import jalaali from 'jalaali-js'
import { usePlanBuilderStore } from '~/stores/planBuilder'
const route = useRoute()
const clientId = route.params.clientId as string
const planId = route.params.planId as string
const store = usePlanBuilderStore()
const showActivateModal = ref(false)
const activating = ref(false)

onMounted(() => store.loadPlan(planId))
onUnmounted(() => store.$reset())

function toShamsi(iso: string) {
  const [y, m, d] = iso.split('-').map(Number)
  const j = jalaali.toJalaali(y, m, d)
  return `${j.jy}/${String(j.jm).padStart(2,'0')}/${String(j.jd).padStart(2,'0')}`
}

async function handleAddDay() {
  if (!store.currentPlan) return
  const dayNumber = store.currentPlan.days.length + 1
  await store.addDay(planId, { day_number: dayNumber })
}

async function handleActivate() {
  activating.value = true
  try {
    await store.activatePlan(planId)
    showActivateModal.value = false
  } catch {
    // error is set in store
  } finally { activating.value = false }
}
</script>

        {label:'جزئیات برنامه'}
      ]" />
    </div>

    <div v-if="store.loading" class="p-4 space-y-3">
      <div v-for="i in 4" :key="i" class="bg-white rounded-2xl p-4 h-20 animate-pulse" />
    </div>
    <template v-else-if="store.currentPlan">
      <div class="p-4 space-y-4">
        <!-- Plan header card -->
        <div class="bg-white rounded-2xl p-4 shadow-sm">
          <div class="flex items-center justify-between gap-2 mb-3">
            <StatusBadge :status="store.currentPlan.status" />
            <NuxtLink v-if="store.currentPlan.status === 'draft'" :to="`/nutritionist/clients/${clientId}/plans/${planId}/edit`"
              class="text-sm text-emerald-600 font-medium">ویرایش</NuxtLink>
          </div>
          <p class="text-sm text-gray-600">
            از {{ toShamsi(store.currentPlan.start_date) }}
            <span v-if="store.currentPlan.end_date"> تا {{ toShamsi(store.currentPlan.end_date) }}</span>

              :editable="store.currentPlan.status === 'draft'"
              @delete="store.deleteDay(planId, $event)" />
          </div>
          <button v-if="store.currentPlan.status === 'draft'" @click="handleAddDay"
            class="mt-3 w-full border-2 border-dashed border-emerald-300 text-emerald-600 rounded-2xl py-4 text-sm font-medium hover:bg-emerald-50 transition-colors">
            + افزودن روز

        <!-- Medications section -->
        <div v-if="store.currentPlan.medications.length">

<script setup lang="ts">
import jalaali from 'jalaali-js'
import { usePlanBuilderStore } from '~/stores/planBuilder'
const route = useRoute()
const router = useRouter()
const clientId = route.params.clientId as string
const planId = route.params.planId as string
const store = usePlanBuilderStore()
const submitting = ref(false)
const error = ref<string | null>(null)

const form = reactive({ startDateShamsi: '', endDateShamsi: '', notes: '', waterMl: 2000 })

onMounted(async () => {
  await store.loadPlan(planId)
  if (store.currentPlan) {
    const { currentPlan: p } = store
    const start = p.start_date.split('-').map(Number)
    const js = jalaali.toJalaali(start[0], start[1], start[2])
    form.startDateShamsi = `${js.jy}/${String(js.jm).padStart(2,'0')}/${String(js.jd).padStart(2,'0')}`
    if (p.end_date) {
      const end = p.end_date.split('-').map(Number)
      const je = jalaali.toJalaali(end[0], end[1], end[2])
      form.endDateShamsi = `${je.jy}/${String(je.jm).padStart(2,'0')}/${String(je.jd).padStart(2,'0')}`
    }
    form.notes = p.notes ?? ''
    form.waterMl = p.daily_water_target_ml ?? 2000
  }
})

function parseShamsi(input: string): string | null {
  const normalized = input.replace(/[۰-۹]/g, d => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)))
  const parts = normalized.split('/')
  if (parts.length !== 3) return null
  const [jy, jm, jd] = parts.map(Number)
  if (isNaN(jy) || isNaN(jm) || isNaN(jd)) return null
  const { gy, gm, gd } = jalaali.toGregorian(jy, jm, jd)
  return `${gy}-${String(gm).padStart(2,'0')}-${String(gd).padStart(2,'0')}`
}

async function onSubmit() {
  error.value = null
  const startDate = parseShamsi(form.startDateShamsi)
  if (!startDate) { error.value = 'تاریخ شروع معتبر نیست'; return }
  const endDate = form.endDateShamsi ? parseShamsi(form.endDateShamsi) : undefined
  submitting.value = true
  try {
    await store.updatePlanHeader(planId, { start_date: startDate, end_date: endDate, notes: form.notes || undefined, daily_water_target_ml: form.waterMl || undefined })
    router.push(`/nutritionist/clients/${clientId}/plans/${planId}`)
  } catch (e: any) {
    error.value = e.data?.error ?? store.error ?? 'خطا'
  } finally { submitting.value = false }
}
</script>

    </div>
    <form v-else @submit.prevent="onSubmit" class="p-4 space-y-4 max-w-lg mx-auto">
      <div v-if="error" class="bg-red-50 text-red-700 rounded-xl px-4 py-3 text-sm">{{ error }}</div>

        <textarea v-model="form.notes" rows="3"
          class="w-full bg-white border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500 resize-none" />
      </div>
      <div>
        <label class="block text-sm font-medium text-gray-700 mb-1">هدف آب روزانه (میلی‌لیتر)</label>
