**1. `OptionCard.vue`** — Accordion option card

Props: `option: MealOptionResponse`, `planId: string`, `dayId: string`, `mealId: string`, `editable?: boolean`
Emits: `delete: [id: string]`, `open-picker: [optionId: string]`

Accordion: `const isOpen = ref(true)` — starts open. Toggle via header click.
**CRITICAL: use `v-show` for body, NOT `v-if`** — v-if destroys reactive state.

```vue
<script setup lang="ts">
import type { MealOptionResponse } from '~/types/plan.types'
import { useNutritionComputed } from '~/composables/useNutritionComputed'
const props = defineProps<{
  option: MealOptionResponse
  planId: string
  dayId: string
  mealId: string
  editable?: boolean
}>()
const emit = defineEmits<{
  delete: [id: string]
  'open-picker': [optionId: string]
  'delete-item': [optionId: string, itemId: string]
  'update-item': [optionId: string, itemId: string, quantity: number]
}>()
const { optionTotals } = useNutritionComputed()
const isOpen = ref(true)
const totals = computed(() => optionTotals(props.option))
const optionLabel = computed(() => props.option.label ? `گزینه ${props.option.option_number} — ${props.option.label}` : `گزینه ${props.option.option_number}`)
</script>
<template>
  <div class="bg-white rounded-2xl shadow-sm overflow-hidden">
    <!-- Accordion header -->
    <button @click="isOpen = !isOpen"
      class="w-full flex items-center justify-between p-4 text-start hover:bg-gray-50 transition-colors"
      :aria-expanded="isOpen" :aria-controls="`option-${option.id}`">
      <div>
        <span class="font-semibold text-gray-800 text-sm">{{ optionLabel }}</span>
        <div class="mt-1">
          <NutritionBadges :totals="totals" compact />
        </div>
      </div>
      <div class="flex items-center gap-2 shrink-0">
        <button v-if="editable" @click.stop="emit('delete', option.id)"
          class="text-xs text-red-400 hover:text-red-600 transition-colors"
          :aria-label="`حذف ${optionLabel}`">حذف</button>
        <span class="text-gray-400 transition-transform" :class="{ 'rotate-180': isOpen }">▼</span>
      </div>
    </button>

    <!-- Accordion body — v-show REQUIRED (not v-if) -->
    <div v-show="isOpen" :id="`option-${option.id}`" class="px-4 pb-4 space-y-2">
      <FoodItemRow v-for="item in option.items" :key="item.id"
        :item="item" :editable="editable"
        @delete="emit('delete-item', option.id, item.id)" />
      <button v-if="editable" @click="emit('open-picker', option.id)"
        class="w-full border border-dashed border-emerald-300 text-emerald-600 rounded-xl py-3 text-sm hover:bg-emerald-50 transition-colors">
        + افزودن غذا
      </button>
      <p v-if="!option.items.length && !editable" class="text-xs text-gray-400 text-center py-2">بدون آیتم</p>
    </div>
  </div>
</template>
```

**3. `FoodPickerSheet.vue`** — Bottom sheet with food search

Props: `open: boolean`, `optionId: string | null`, `planId: string`, `dayId: string`, `mealId: string`
Emits: `close`, `item-added`

This component manages its own local search state + debounce + quantity form.
**Uses Teleport to body** for proper fixed positioning.
**300ms slide-up transition** via custom CSS.

```vue
<script setup lang="ts">
import type { FoodEmbedded } from '~/types/plan.types'
const props = defineProps<{
  open: boolean
  optionId: string | null
  planId: string
  dayId: string
  mealId: string
}>()
const emit = defineEmits<{ close: []; 'item-added': [] }>()

// Search state
const searchQuery = ref('')
const searchResults = ref<FoodEmbedded[]>([])
const searchLoading = ref(false)
const selectedFood = ref<FoodEmbedded | null>(null)
const quantity = ref<number>(100)
const addingItem = ref(false)
const addError = ref<string | null>(null)

// Debounced search
let debounceTimer: ReturnType<typeof setTimeout> | null = null
watch(searchQuery, (q) => {
  if (debounceTimer) clearTimeout(debounceTimer)
  if (!q.trim()) { searchResults.value = []; return }
  debounceTimer = setTimeout(() => searchFoods(q), 300)
})

async function searchFoods(q: string) {
  searchLoading.value = true
  try {
    const data = await $fetch<{ foods: FoodEmbedded[] }>(`/api/foods?search=${encodeURIComponent(q)}&limit=20`)
    searchResults.value = data.foods ?? []
  } catch { searchResults.value = [] }
  finally { searchLoading.value = false }
}

function selectFood(food: FoodEmbedded) {
  selectedFood.value = food
  quantity.value = food.measurement_amount  // default to 1 serving
  addError.value = null
}

async function addItem() {
  if (!selectedFood.value || !props.optionId) return
  if (quantity.value <= 0) { addError.value = 'مقدار باید بیشتر از صفر باشد'; return }
  addingItem.value = true
  addError.value = null
  try {
    await $fetch(`/api/diet-plans/${props.planId}/days/${props.dayId}/meals/${props.mealId}/options/${props.optionId}/items`, {
      method: 'POST',
      body: { food_id: selectedFood.value.id, quantity: quantity.value, measurement_unit: selectedFood.value.measurement_unit }
    })
    emit('item-added')
    handleClose()
  } catch (e: any) {
    addError.value = e.data?.error ?? 'خطا در افزودن آیتم'
  } finally { addingItem.value = false }
}

function handleClose() {
  searchQuery.value = ''
  searchResults.value = []
  selectedFood.value = null
  quantity.value = 100
  addError.value = null
  emit('close')
}

// Reset search when sheet opens
watch(() => props.open, (newVal) => {
  if (!newVal) return
  searchQuery.value = ''
  searchResults.value = []
  selectedFood.value = null
})
</script>
<template>
  <Teleport to="body">
    <Transition name="slide-up">
      <div v-if="open" class="fixed inset-0 z-50 flex items-end justify-center"
        role="dialog" aria-modal="true" aria-label="انتخاب غذا">
        <!-- Backdrop -->
        <div class="absolute inset-0 bg-black/40" @click="handleClose" />
        <!-- Sheet -->
        <div class="relative bg-white rounded-t-3xl w-full max-w-lg max-h-[80vh] flex flex-col shadow-2xl">
          <!-- Handle bar -->
          <div class="flex justify-center pt-3 pb-2">
            <div class="w-12 h-1 bg-gray-200 rounded-full" />
          </div>
          <!-- Header -->
          <div class="px-4 pb-3 flex items-center justify-between border-b border-gray-100">
            <h2 class="text-base font-bold text-gray-800">انتخاب غذا</h2>
            <button @click="handleClose" class="text-gray-400 hover:text-gray-600 text-sm">بستن</button>
          </div>

          <template v-if="!selectedFood">
            <!-- Search input -->
            <div class="px-4 py-3">
              <input v-model="searchQuery" type="search" placeholder="جستجوی غذا..."
                class="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500"
                aria-label="جستجوی غذا" autofocus />
            </div>
            <!-- Results list -->
            <div class="overflow-y-auto flex-1 px-4 pb-4">
              <div v-if="searchLoading" class="space-y-2">
                <div v-for="i in 4" :key="i" class="h-14 bg-gray-50 rounded-xl animate-pulse" />
              </div>
              <div v-else-if="!searchQuery.trim()" class="text-center text-gray-400 py-8 text-sm">
                نام غذا را وارد کنید
              </div>
              <div v-else-if="!searchResults.length" class="text-center text-gray-400 py-8 text-sm">
                نتیجه‌ای یافت نشد
              </div>
              <button v-else v-for="food in searchResults" :key="food.id"
                @click="selectFood(food)"
                class="w-full text-start flex items-center justify-between gap-3 py-3 border-b border-gray-50 last:border-0 hover:bg-gray-50 transition-colors rounded-lg px-2">
                <div>
                  <p class="text-sm font-medium text-gray-800">{{ food.name }}</p>
                  <p class="text-xs text-gray-500 mt-0.5">
                    {{ food.measurement_amount }} {{ food.measurement_unit }} — {{ food.calories }} کیلوکالری
                  </p>
                </div>
                <span class="text-emerald-600 text-lg shrink-0">+</span>
              </button>
            </div>
          </template>

          <!-- Quantity form after food selected -->
          <template v-else>
            <div class="px-4 py-4 flex-1">
              <div class="mb-4">
                <p class="font-semibold text-gray-800">{{ selectedFood.name }}</p>
                <p class="text-xs text-gray-500 mt-1">
                  هر {{ selectedFood.measurement_amount }} {{ selectedFood.measurement_unit }} = {{ selectedFood.calories }} کیلوکالری
                </p>
              </div>
              <div>
                <label class="block text-sm font-medium text-gray-700 mb-1">
                  مقدار ({{ selectedFood.measurement_unit }})
                </label>
                <input v-model.number="quantity" type="number" min="1" step="1"
                  class="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500" />
              </div>
              <div v-if="addError" class="mt-2 text-red-600 text-xs">{{ addError }}</div>
              <div class="mt-4 flex gap-3">
                <button @click="addItem" :disabled="addingItem"
                  class="flex-1 bg-emerald-600 text-white py-3 rounded-xl font-medium hover:bg-emerald-700 disabled:opacity-50 transition-colors">
                  {{ addingItem ? 'در حال افزودن...' : 'افزودن' }}
                </button>
                <button @click="selectedFood = null"
                  class="flex-1 bg-gray-100 text-gray-700 py-3 rounded-xl hover:bg-gray-200 transition-colors">
                  برگشت
                </button>
              </div>
            </div>
          </template>
        </div>
      </div>
    </Transition>
  </Teleport>
</template>

}

}
```

Create the meal view page. This is the deepest level of the drill-down — the user sees all options for a meal, can manage options and food items, and opens FoodPickerSheet to add items.

Route params: clientId, planId, dayId, mealId

Data source: `store.currentPlan.days.find(d => d.id === dayId).meals.find(m => m.id === mealId)`

```vue
<script setup lang="ts">
import { usePlanBuilderStore } from '~/stores/planBuilder'
import { useNutritionComputed } from '~/composables/useNutritionComputed'
const route = useRoute()
const clientId = route.params.clientId as string
const planId = route.params.planId as string
const dayId = route.params.dayId as string
const mealId = route.params.mealId as string
const store = usePlanBuilderStore()
const { mealTotals } = useNutritionComputed()

onMounted(() => { if (!store.currentPlan) store.loadPlan(planId) })
onUnmounted(() => store.$reset())

const meal = computed(() => {
  const day = store.currentPlan?.days.find(d => d.id === dayId)
  return day?.meals.find(m => m.id === mealId) ?? null
})
const isDraft = computed(() => store.currentPlan?.status === 'draft')
const mealNutrition = computed(() => meal.value ? mealTotals(meal.value) : { calories: 0, protein_g: 0, carbs_g: 0, fat_g: 0, fiber_g: 0 })

// FoodPickerSheet state delegated to store (D-21)
const pickerOpen = computed(() => store.foodPicker.open)
const pickerOptionId = computed(() => store.foodPicker.targetOptionId)

async function onItemAdded() {
  // Picker closes itself; refresh the aggregate to reflect new item with food data
  await store.loadPlan(planId)
}
</script>
<template>
  <div class="min-h-screen bg-gray-50 pb-24" dir="rtl">
    <!-- Sticky top bar with meal title and live nutrition -->
    <div class="sticky top-0 bg-white z-10 px-4 py-3 border-b border-gray-100">
      <PlanBreadcrumb :items="[
        {label:'برنامه', to:`/nutritionist/clients/${clientId}/plans/${planId}`},
        {label:'روز', to:`/nutritionist/clients/${clientId}/plans/${planId}/days/${dayId}`},
        {label: meal?.title ?? '...'}
      ]" />
      <div v-if="meal" class="mt-2">
        <NutritionBadges :totals="mealNutrition" />
      </div>
    </div>

    <div v-if="store.loading" class="p-4 space-y-3">
      <div v-for="i in 2" :key="i" class="bg-white rounded-2xl p-4 h-32 animate-pulse" />
    </div>
    <template v-else-if="meal">
      <div class="p-4 space-y-4">
        <!-- Options list -->
        <div>
          <h2 class="text-base font-bold text-gray-800 mb-3">گزینه‌ها</h2>
          <div class="space-y-3">
            <OptionCard v-for="option in meal.options" :key="option.id"
              :option="option"
              :planId="planId" :dayId="dayId" :mealId="mealId"
              :editable="isDraft"
              @delete="store.deleteOption(planId, dayId, mealId, option.id)"
              @delete-item="(optId, itemId) => store.deleteItem(planId, dayId, mealId, optId, itemId)"
              @open-picker="store.openFoodPicker($event)" />
          </div>
          <button v-if="isDraft" @click="store.addOption(planId, dayId, mealId)"
            class="mt-3 w-full border-2 border-dashed border-emerald-300 text-emerald-600 rounded-2xl py-4 text-sm font-medium hover:bg-emerald-50 transition-colors">
            + افزودن گزینه
          </button>
        </div>
      </div>
    </template>
    <div v-else class="p-4 text-center text-gray-400 py-12">وعده یافت نشد</div>

    <!-- FoodPickerSheet — rendered once, controlled by store -->
    <FoodPickerSheet
      :open="pickerOpen"
      :optionId="pickerOptionId"
      :planId="planId"
      :dayId="dayId"
      :mealId="mealId"
      @close="store.closeFoodPicker()"
      @item-added="onItemAdded()" />
  </div>
</template>
```
