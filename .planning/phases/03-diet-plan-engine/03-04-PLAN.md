import type { DietPlanResponse, FoodPickerState } from '~/types/plan.types'

export const usePlanBuilderStore = defineStore('planBuilder', () => {
  const currentPlan = ref<DietPlanResponse | null>(null)
  const loading = ref(false)
  const error = ref<string | null>(null)

  const foodPicker = ref<FoodPickerState>({
    open: false,
    targetOptionId: null,
    searchQuery: '',
    searchResults: [],
    loading: false
  })

  // Core load action — called after EVERY CRUD mutation (D-22/Pitfall 6)
  async function loadPlan(planId: string) {
    loading.value = true
    error.value = null
    try {
      const data = await $fetch<DietPlanResponse>(`/api/diet-plans/${planId}`)
      currentPlan.value = data
    } catch (e: any) {
      error.value = e.data?.error ?? 'خطا در بارگذاری برنامه'
    } finally {
      loading.value = false
    }
  }

  // Plan header mutations — call loadPlan after each
  async function createPlan(clientId: string, body: object) {
    return await $fetch<DietPlanResponse>('/api/diet-plans', { method: 'POST', body: { client_id: clientId, ...body } })
  }

  async function updatePlanHeader(planId: string, body: object) {
    await $fetch(`/api/diet-plans/${planId}`, { method: 'PATCH', body })
    await loadPlan(planId)

    await $fetch(`/api/diet-plans/${planId}/activate`, { method: 'PATCH' })
    await loadPlan(planId)

import { defineStore } from 'pinia'
import dayjs from 'dayjs'
import type { DietPlanResponse } from '~/types/plan.types'

export const useClientPlanStore = defineStore('clientPlan', () => {
  const activePlan = ref<DietPlanResponse | null>(null)
  const activeDayNumber = ref<number>(1)
  const loading = ref(false)
  const error = ref<string | null>(null)

  async function fetchActivePlan() {
    loading.value = true
    error.value = null
    try {
      const data = await $fetch<DietPlanResponse>('/api/clients/me/active-plan')
      activePlan.value = data
      initActiveDay()
    } catch (e: any) {
      if (e.status === 404) {
        activePlan.value = null
      } else {
        error.value = e.data?.error ?? 'خطا در بارگذاری برنامه'
      }
    } finally {
      loading.value = false
    }
  }

  function initActiveDay() {
    if (!activePlan.value) { activeDayNumber.value = 1; return }
    const startDate = dayjs(activePlan.value.start_date)
    const today = dayjs()
    const diffDays = today.diff(startDate, 'day') + 1  // day_number is 1-indexed
    const totalDays = activePlan.value.days.length
    // Clamp to valid day range
    if (diffDays < 1) {
      activeDayNumber.value = 1
    } else if (diffDays > totalDays) {
      activeDayNumber.value = totalDays
    } else {
      activeDayNumber.value = diffDays
    }
  }

  const activeDay = computed(() => {
    if (!activePlan.value) return null
    return activePlan.value.days.find(d => d.day_number === activeDayNumber.value) ?? null
  })

  function setActiveDay(dayNumber: number) {
    activeDayNumber.value = dayNumber
  }

  return { activePlan, activeDayNumber, activeDay, loading, error, fetchActivePlan, initActiveDay, setActiveDay }
})

import type { MealOptionItemResponse, MealOptionResponse, MealResponse, PlanDayResponse, NutritionTotals } from '~/types/plan.types'

export function useNutritionComputed() {
  // item.quantity / item.food.measurement_amount is the serving ratio (D-14)
  function itemTotals(item: MealOptionItemResponse): NutritionTotals {
    const ratio = item.food.measurement_amount > 0 ? item.quantity / item.food.measurement_amount : 0
    return {
      calories: +(item.food.calories * ratio).toFixed(1),
      protein_g: +(item.food.protein_g * ratio).toFixed(1),
      carbs_g: +(item.food.carbs_g * ratio).toFixed(1),
      fat_g: +(item.food.fat_g * ratio).toFixed(1),
      fiber_g: +(item.food.fiber_g * ratio).toFixed(1),
    }
  }

  function sumTotals(totals: NutritionTotals[]): NutritionTotals {
    return totals.reduce((acc, t) => ({
      calories: +(acc.calories + t.calories).toFixed(1),
      protein_g: +(acc.protein_g + t.protein_g).toFixed(1),
      carbs_g: +(acc.carbs_g + t.carbs_g).toFixed(1),
      fat_g: +(acc.fat_g + t.fat_g).toFixed(1),
      fiber_g: +(acc.fiber_g + t.fiber_g).toFixed(1),
    }), { calories: 0, protein_g: 0, carbs_g: 0, fat_g: 0, fiber_g: 0 })
  }

  // Best option totals: use option with highest calories as representative (or option 1)
  function optionTotals(option: MealOptionResponse): NutritionTotals {
    return sumTotals(option.items.map(itemTotals))
  }

  function mealTotals(meal: MealResponse): NutritionTotals {
    if (!meal.options.length) return { calories: 0, protein_g: 0, carbs_g: 0, fat_g: 0, fiber_g: 0 }
    // Use first option (option 1) as representative for meal totals (per UI-SPEC)
    return optionTotals(meal.options[0])
  }

  function dayTotals(day: PlanDayResponse): NutritionTotals {
    return sumTotals(day.meals.map(mealTotals))
  }

  return { itemTotals, optionTotals, mealTotals, dayTotals }
}
