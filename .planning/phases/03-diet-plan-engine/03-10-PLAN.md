**Step 1: Add sqlc query to diet_plans.sql**

Append the following named query to `backend/db/queries/diet_plans.sql` (after existing queries):

```sql
-- name: ListClientOwnPlans :many
-- Returns all plans for the authenticated client ordered by creation date (newest first).
-- Used by GET /api/clients/me/plans for client plan history tab.
SELECT
    dp.id,
    dp.status,
    dp.start_date,
    dp.end_date,
    dp.created_at,
    COUNT(pd.id)::int AS day_count
FROM diet_plans dp
LEFT JOIN plan_days pd ON pd.plan_id = dp.id
WHERE dp.client_id = $1
GROUP BY dp.id, dp.status, dp.start_date, dp.end_date, dp.created_at
ORDER BY dp.created_at DESC
LIMIT sqlc.arg(limit_val)::int OFFSET sqlc.arg(offset_val)::int;

-- name: CountClientOwnPlans :one
SELECT COUNT(*) FROM diet_plans WHERE client_id = $1;
```

**Step 3: Add ListMyPlans handler to diet_plan_handler.go**

Add the following method to the existing `DietPlanHandler` struct in `backend/internal/handler/diet_plan_handler.go`:

```go
// ListMyPlans handles GET /api/clients/me/plans
// Returns paginated list of all plans (active + archived + draft) for the authenticated client.
// Row-level security: client_id derived from JWT, never from request body/query.
func (h *DietPlanHandler) ListMyPlans(c *gin.Context) {
    clientID := middleware.GetUserID(c) // from JWT — enforces client isolation

    page := 1
    if p, err := strconv.Atoi(c.DefaultQuery("page", "1")); err == nil && p > 0 {
        page = p
    }
    const limit = 20
    offset := (page - 1) * limit

    plans, err := h.svc.ListClientOwnPlans(c.Request.Context(), clientID, limit, offset)
    if err != nil {
        c.JSON(http.StatusInternalServerError, dto.ErrorResponse{Error: "خطا در دریافت تاریخچه برنامه‌ها"})
        return
    }
    total, err := h.svc.CountClientOwnPlans(c.Request.Context(), clientID)
    if err != nil {
        total = int64(len(plans))
    }

    c.JSON(http.StatusOK, dto.DietPlanListResponse{
        Data:    plans,
        Total:   total,
        Page:    page,
        Limit:   limit,
        HasMore: int64(page*limit) < total,
    })
}
```
