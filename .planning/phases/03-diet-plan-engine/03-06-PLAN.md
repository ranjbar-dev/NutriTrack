Reorder logic — swap display_order values between adjacent meals:
```typescript
async function moveUp(mealId: string) {
  if (!day.value) return
  const meals = [...day.value.meals].sort((a, b) => a.display_order - b.display_order)
  const idx = meals.findIndex(m => m.id === mealId)
  if (idx <= 0) return
  // Swap orders: current meal gets prev meal's order
  await store.reorderMeal(planId, dayId, mealId, meals[idx - 1].display_order)
  await store.reorderMeal(planId, dayId, meals[idx - 1].id, meals[idx].display_order)
  // loadPlan is called inside each reorderMeal call — second call syncs final state
}
async function moveDown(mealId: string) {
  if (!day.value) return
  const meals = [...day.value.meals].sort((a, b) => a.display_order - b.display_order)
  const idx = meals.findIndex(m => m.id === mealId)
  if (idx >= meals.length - 1) return
  await store.reorderMeal(planId, dayId, mealId, meals[idx + 1].display_order)
  await store.reorderMeal(planId, dayId, meals[idx + 1].id, meals[idx].display_order)
}
```

**2. `days/[dayId]/meals/new.vue`** — Add Meal Form:

Simple form: meal title (required), scheduled_time (optional time input HH:MM), display_order (hidden, auto-computed as meals.length + 1).
On submit: `store.addMeal(planId, dayId, body)` → on success navigate to the new meal's view.

Note: store.addMeal returns void and calls loadPlan internally. We need the meal ID to navigate to it. Solution: call $fetch directly to get the new meal ID, then navigate, and optionally call loadPlan.

```vue
<script setup lang="ts">
import { usePlanBuilderStore } from '~/stores/planBuilder'
import type { MealResponse } from '~/types/plan.types'
const route = useRoute()
const router = useRouter()
const clientId = route.params.clientId as string
const planId = route.params.planId as string
const dayId = route.params.dayId as string
const store = usePlanBuilderStore()
const submitting = ref(false)
const error = ref<string | null>(null)
const form = reactive({ title: '', scheduledTime: '' })

// Get display_order from store (meals.length + 1)
const nextDisplayOrder = computed(() => {
  const day = store.currentPlan?.days.find(d => d.id === dayId)
  return (day?.meals.length ?? 0) + 1
})

onMounted(() => { if (!store.currentPlan) store.loadPlan(planId) })

async function onSubmit() {
  if (!form.title.trim()) { error.value = 'عنوان وعده الزامی است'; return }
  error.value = null
  submitting.value = true
  try {
    // Direct $fetch to get the new meal ID for navigation
    const meal = await $fetch<MealResponse>(`/api/diet-plans/${planId}/days/${dayId}/meals`, {
      method: 'POST',
      body: {
        title: form.title,
        scheduled_time: form.scheduledTime || undefined,
        display_order: nextDisplayOrder.value,
      }
    })
    // Refresh store
    await store.loadPlan(planId)
    router.push(`/nutritionist/clients/${clientId}/plans/${planId}/days/${dayId}/meals/${meal.id}`)
  } catch (e: any) {
    error.value = e.data?.error ?? 'خطا در افزودن وعده'
  } finally { submitting.value = false }
}
</script>
<template>
  <div class="min-h-screen bg-gray-50" dir="rtl">
    <div class="sticky top-0 bg-white z-10 px-4 py-3 border-b border-gray-100">
      <PlanBreadcrumb :items="[
        {label:'برنامه', to:`/nutritionist/clients/${clientId}/plans/${planId}`},
        {label:`روز`, to:`/nutritionist/clients/${clientId}/plans/${planId}/days/${dayId}`},
        {label:'وعده جدید'}
      ]" />
    </div>
    <form @submit.prevent="onSubmit" class="p-4 space-y-4 max-w-lg mx-auto">
      <div v-if="error" class="bg-red-50 text-red-700 rounded-xl px-4 py-3 text-sm">{{ error }}</div>
      <div>
        <label class="block text-sm font-medium text-gray-700 mb-1">عنوان وعده <span class="text-red-500">*</span></label>
        <input v-model="form.title" type="text" placeholder="مثال: صبحانه"
          class="w-full bg-white border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500" />
      </div>
      <div>
        <label class="block text-sm font-medium text-gray-700 mb-1">ساعت وعده (اختیاری)</label>
        <input v-model="form.scheduledTime" type="time"
          class="w-full bg-white border border-gray-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500" />
      </div>
      <button type="submit" :disabled="submitting"
        class="w-full bg-emerald-600 text-white py-3 rounded-xl font-medium hover:bg-emerald-700 disabled:opacity-50 transition-colors">
        {{ submitting ? 'در حال ذخیره...' : 'افزودن وعده' }}
      </button>
    </form>
  </div>
</template>
```
