Create `frontend/app/components/plan/DayTabBar.vue`.

Props: `days: PlanDayResponse[]`, `activeDayNumber: number`
Emits: `update:activeDayNumber: [number]`

Layout: horizontal scrollable flex container. Each tab is a button showing "روز N" (+ label if present).
Active tab: `bg-emerald-600 text-white`, inactive: `bg-white text-gray-600 border border-gray-200`.

Auto-scroll: when `activeDayNumber` changes, scroll the active tab button into center view.

```vue
<script setup lang="ts">
import type { PlanDayResponse } from '~/types/plan.types'
const props = defineProps<{
  days: PlanDayResponse[]
  activeDayNumber: number
}>()
const emit = defineEmits<{ 'update:activeDayNumber': [number] }>()

const tabRefs = ref<HTMLButtonElement[]>([])
const scrollContainer = ref<HTMLDivElement | null>(null)

function setTab(dayNumber: number) {
  emit('update:activeDayNumber', dayNumber)
}

// Auto-scroll active tab into center view
watch(() => props.activeDayNumber, (dayNum) => {
  nextTick(() => {
    const idx = props.days.findIndex(d => d.day_number === dayNum)
    if (idx >= 0 && tabRefs.value[idx]) {
      tabRefs.value[idx].scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' })
    }
  })
}, { immediate: true })
</script>
<template>
  <div ref="scrollContainer"
    class="flex gap-2 overflow-x-auto scroll-smooth scrollbar-hide py-2 px-4 bg-white border-b border-gray-100"
    role="tablist" aria-label="روزهای برنامه">
    <button
      v-for="(day, idx) in days"
      :key="day.id"
      :ref="el => { if (el) tabRefs[idx] = el as HTMLButtonElement }"
      @click="setTab(day.day_number)"
      role="tab"
      :aria-selected="day.day_number === activeDayNumber"
      :class="[
        'shrink-0 px-4 py-2 rounded-full text-sm font-medium transition-colors whitespace-nowrap',
        day.day_number === activeDayNumber
          ? 'bg-emerald-600 text-white'
          : 'bg-white text-gray-600 border border-gray-200 hover:border-emerald-300'
      ]"
    >
      روز {{ day.day_number }}<span v-if="day.label"> — {{ day.label }}</span>
    </button>
  </div>
</template>
<style scoped>
.scrollbar-hide::-webkit-scrollbar { display: none; }
.scrollbar-hide { -ms-overflow-style: none; scrollbar-width: none; }
</style>
```

```vue
<script setup lang="ts">
import { useClientPlanStore } from '~/stores/clientPlan'
import { useNutritionComputed } from '~/composables/useNutritionComputed'
const store = useClientPlanStore()
const { dayTotals, mealTotals, optionTotals } = useNutritionComputed()

onMounted(() => store.fetchActivePlan())

const activeDayTotals = computed(() =>
  store.activeDay ? dayTotals(store.activeDay) : { calories: 0, protein_g: 0, carbs_g: 0, fat_g: 0, fiber_g: 0 }
)
</script>
<template>
  <div class="min-h-screen bg-gray-50" dir="rtl">
    <!-- Loading skeleton -->
    <template v-if="store.loading">
      <div class="p-4 pt-6 space-y-3">
        <div class="h-8 bg-gray-100 rounded-xl animate-pulse w-1/2" />
        <div class="h-10 bg-gray-100 rounded-full animate-pulse" />
        <div v-for="i in 3" :key="i" class="h-24 bg-gray-100 rounded-2xl animate-pulse" />
      </div>
    </template>

    <!-- Empty state -->
    <template v-else-if="!store.activePlan">
      <div class="flex flex-col items-center justify-center py-24 px-4 text-center">
        <div class="text-5xl mb-4">📋</div>
        <h2 class="text-lg font-bold text-gray-700 mb-2">برنامه‌ای فعال ندارید</h2>
        <p class="text-sm text-gray-400">متخصص تغذیه شما هنوز برنامه‌ای برای شما فعال نکرده است</p>
      </div>
    </template>

    <!-- Active plan view -->
    <template v-else>
      <!-- Sticky plan header -->
      <div class="sticky top-0 bg-white z-20 px-4 py-3 border-b border-gray-100">
        <div class="flex items-center justify-between gap-2">
          <h1 class="text-lg font-bold text-gray-800">برنامه من</h1>
          <NutritionBadges v-if="store.activeDay" :totals="activeDayTotals" compact />
        </div>
      </div>

      <!-- Sticky DayTabBar below header -->
      <div class="sticky top-[56px] z-10">
        <DayTabBar
          :days="store.activePlan.days"
          :activeDayNumber="store.activeDayNumber"
          @update:activeDayNumber="store.setActiveDay($event)" />
      </div>

      <!-- Day content (day-specific: meals + exercises only — changes when day tab changes) -->
      <div v-if="store.activeDay" class="p-4 space-y-4">
        <div v-for="meal in store.activeDay.meals" :key="meal.id">
          <div class="mb-2">
            <div class="flex items-center justify-between gap-2">
              <h3 class="font-semibold text-gray-800">{{ meal.title }}</h3>
              <span v-if="meal.scheduled_time"
                class="text-xs bg-gray-100 text-gray-600 px-2 py-0.5 rounded-full">
                {{ meal.scheduled_time.slice(0, 5) }}
              </span>
            </div>
          </div>
          <!-- Options (read-only) -->
          <div class="space-y-2">
            <div v-for="option in meal.options" :key="option.id"
              class="bg-white rounded-2xl p-4 shadow-sm">
              <div class="flex items-center justify-between gap-2 mb-3">
                <span class="text-sm font-medium text-gray-700">
                  گزینه {{ option.option_number }}<span v-if="option.label"> — {{ option.label }}</span>
                </span>
                <NutritionBadges :totals="optionTotals(option)" compact />
              </div>
              <div class="space-y-1">
                <FoodItemRow v-for="item in option.items" :key="item.id"
                  :item="item" :editable="false" />
              </div>
            </div>
          </div>
        </div>

        <!-- Exercises section (day-specific) -->
        <div v-if="store.activeDay.exercises.length">
          <h2 class="text-base font-bold text-gray-800 mb-3">تمرین‌ها</h2>
          <div class="space-y-2">
            <ExerciseCard v-for="ex in store.activeDay.exercises" :key="ex.id"
              :exercise="ex" :editable="false" />
          </div>
        </div>
      </div>

      <!-- Plan-level sections: medications + water target.
           D-27: medications shown ONCE at plan level — NOT inside the day-switching block above.
           Rendered from store.activePlan (not store.activeDay) so they never repeat per-day. -->
      <div class="px-4 space-y-4 pb-24">
        <!-- WaterBadge (plan-level) -->
        <WaterBadge v-if="store.activePlan.daily_water_target_ml"
          :ml="store.activePlan.daily_water_target_ml" />

        <!-- Medications (plan-level, D-27: dedicated card, not per-day) -->
        <div v-if="store.activePlan.medications.length">
          <h2 class="text-base font-bold text-gray-800 mb-3">داروها</h2>
          <div class="space-y-2">
            <MedicationCard v-for="med in store.activePlan.medications" :key="med.id"
              :medication="med" :editable="false" />
          </div>
        </div>
      </div>
    </template>
  </div>
</template>
```
